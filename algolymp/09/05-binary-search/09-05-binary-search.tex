\section{Căutarea binară a unui element}

\begin{frame}{Motivație}
  \textbf{Problemă:} într-un șir \emph{sortat} cu $n$ elemente trebuie aflat dacă valoarea $x$ se află în el.
  \begin{itemize}
    \item \textbf{Căutare secvențială:} elementele se parcurg pe rând $\rightarrow$ în cel mai rău caz, $n$ comparații
  \end{itemize}
  \begin{intrebare}
    Cum poate fi folosit faptul că șirul este sortat pentru a face mai puține comparații?
  \end{intrebare}
  \begin{itemize}
    \item<2-> \textbf{Căutare binară:} se compară $x$ cu elementul din mijloc, iar jumătatea în care $x$ nu se poate afla se elimină la fiecare pas
  \end{itemize}
  \uncover<3->{
    \begin{center}
    \begin{tabular}{lcc}
      \toprule
      $n$ & Secvențial & Binar \\
      \midrule
      $10$            & $10$ pași       & $4$ pași  \\
      $100$           & $100$ pași      & $7$ pași  \\
      $1\,000\,000$   & $10^6$ pași     & $20$ pași \\
      \bottomrule
    \end{tabular}
    \end{center}}
\end{frame}

\begin{frame}{Idee}
  \begin{block}{Condiție}
    Șirul trebuie să fie \textbf{sortat} (crescător sau descrescător).
  \end{block}
  Se păstrează un interval $[st, dr]$ care conține răspunsul. La fiecare pas:
  \begin{enumerate}
    \item Se calculează mijlocul: $mid = (st + dr) \ / \ 2$ (împărțire întreagă)
    \item Se compară $v[mid]$ cu $x$:
  \end{enumerate}
  \begin{intrebare}<2->
    Ce se face în fiecare dintre cazurile $v[mid] = x$, $v[mid] < x$ și $v[mid] > x$?
  \end{intrebare}
  \begin{itemize}
    \item<3-> $v[mid] = x$ $\rightarrow$ \textbf{găsit}
    \item<3-> $v[mid] < x$ $\rightarrow$ $st = mid + 1$ \textemdash{} se caută în dreapta
    \item<3-> $v[mid] > x$ $\rightarrow$ $dr = mid - 1$ \textemdash{} se caută în stânga
  \end{itemize}
  \begin{remarca}<4->{Oprire}
    Căutarea se oprește când valoarea este găsită sau când intervalul devine gol ($st > dr$).
  \end{remarca}
\end{frame}

\begin{frame}{Algoritmul pas cu pas}
  Șir sortat $v = [2,\ 5,\ 8,\ 11,\ 14,\ 17,\ 20,\ 23,\ 26,\ 29,\ 32]$, căutăm $x = 23$.
  \begin{intrebare}
    Câți pași sunt necesari pentru a găsi $x = 23$ și ce interval rămâne după fiecare pas?
  \end{intrebare}
  \begin{center}
  \begin{tikzpicture}
    \uncover<2->{\brow{0}{0}{10}{5}{$mid=5$: $17 < 23 \rightarrow st=6$}}
    \uncover<3->{\brow{-0.8}{6}{10}{8}{$mid=8$: $26 > 23 \rightarrow dr=7$}}
    \uncover<4->{\brow{-1.6}{6}{7}{6}{$mid=6$: $20 < 23 \rightarrow st=7$}}
    \uncover<5->{\brow{-2.4}{7}{7}{7}{$mid=7$: $23 = 23$, găsit}}
  \end{tikzpicture}
  \end{center}
  \begin{remarca}<5->{Concluzie}
    Celula colorată este $mid$, iar celulele estompate sunt eliminate: intervalul $[st, dr]$ se înjumătățește la fiecare pas.
  \end{remarca}
\end{frame}

\begin{frame}[fragile]{Căutare secvențială \textemdash{} implementare}
  \begin{lstlisting}
int n, x, v[100005];       // se citesc n, x si v[0..n-1]

int poz = -1;
for (int i = 0; i < n; i++) {
    if (v[i] == x) {
        poz = i;
        break;
    }
}

cout << poz;
  \end{lstlisting}
  \begin{itemize}
    \item Complexitate: $O(n)$ \textemdash{} în cel mai rău caz se parcurge tot șirul
    \item Funcționează pe orice șir, \textbf{sortat sau nu}
  \end{itemize}
\end{frame}

\begin{frame}[fragile]{Căutare binară \textemdash{} implementare}
  \begin{lstlisting}
int n, x, v[100005];       // se citesc n, x si v[0..n-1]

int st = 0, dr = n - 1, poz = -1;
while (st <= dr) {
    int mid = (st + dr) / 2;
    if (v[mid] == x) {
        poz = mid;
        break;
    } else if (v[mid] < x) {
        st = mid + 1;      // cautam in dreapta
    } else {
        dr = mid - 1;      // cautam in stanga
    }
}

cout << poz;
  \end{lstlisting}
  \begin{itemize}
    \item Complexitate: $O(\log n)$
  \end{itemize}
  \begin{intrebare}<2->
    Ce se întâmplă dacă șirul nu este sortat?
  \end{intrebare}
  \uncover<3->{Eliminarea unei jumătăți nu mai este justificată, deci algoritmul poate rata valoarea căutată.}
\end{frame}

\section{Variații}

\begin{frame}{Lower bound și upper bound}
  Căutarea clasică găsește \textbf{o apariție oarecare} a lui $x$. Adesea este nevoie de:
  \begin{itemize}
    \item \textbf{Lower bound:} prima poziție cu $v[poz] \geq x$
    \item \textbf{Upper bound:} prima poziție cu $v[poz] > x$
  \end{itemize}
  \begin{intrebare}
    Pentru $v = [1,\ 2,\ 2,\ 2,\ 3,\ 5]$ și $x = 2$, care sunt cele două poziții și câte apariții are $x$?
  \end{intrebare}
  \begin{uncoverenv}<2->
    \begin{center}
    \begin{tabular}{lcc}
      \toprule
      & Poziție & Valoare \\
      \midrule
      Lower bound ($\geq 2$) & $1$ & $v[1] = 2$ \\
      Upper bound ($> 2$)    & $4$ & $v[4] = 3$ \\
      \bottomrule
    \end{tabular}
    \end{center}
  \end{uncoverenv}
  \begin{remarca}<3->{Numărul de apariții}
    $\text{upper\_bound} - \text{lower\_bound} = 4 - 1 = 3$.
  \end{remarca}
\end{frame}

\begin{frame}[fragile]{Implementare \texttt{lower\_bound}}
  \begin{lstlisting}
// prima pozitie cu v[poz] >= x
// ans = n daca toate elementele sunt < x
int st = 0, dr = n - 1, ans = n;
while (st <= dr) {
    int mid = (st + dr) / 2;
    if (v[mid] >= x) {
        ans = mid;       // candidat, continuam in stanga
        dr = mid - 1;
    } else {
        st = mid + 1;
    }
}
  \end{lstlisting}
  Când $v[mid] \geq x$, poziția se \textbf{salvează}, iar căutarea continuă mai la stânga, după un candidat mai bun.
  \begin{intrebare}<2->
    Ce condiție trebuie modificată pentru \texttt{upper\_bound}?
  \end{intrebare}
  \uncover<3->{Singura diferență: condiția \lstinline|v[mid] > x| în loc de \lstinline|v[mid] >= x|.}
\end{frame}

\begin{frame}[fragile]{Ultima poziție cu $v[poz] < x$ sau $\leq x$}
  \begin{intrebare}
    Cum se obține ultima poziție cu $v[poz] \leq x$, folosind \texttt{upper\_bound}?
  \end{intrebare}
  \uncover<2->{
    \begin{itemize}
      \item Ultima poziție cu $v[poz] < x$ $=$ \texttt{lower\_bound}$(x) - 1$
      \item Ultima poziție cu $v[poz] \leq x$ $=$ \texttt{upper\_bound}$(x) - 1$
    \end{itemize}
    Sau implementarea se adaptează direct, inversând sensul căutării:}
  \begin{uncoverenv}<3->
    \begin{lstlisting}
// ultima pozitie cu v[poz] <= x
// ans = -1 daca toate elementele sunt > x
int st = 0, dr = n - 1, ans = -1;
while (st <= dr) {
    int mid = (st + dr) / 2;
    if (v[mid] <= x) {
        ans = mid;       // candidat, continuam in dreapta
        st = mid + 1;
    } else {
        dr = mid - 1;
    }
}
    \end{lstlisting}
    Pentru $< x$, condiția devine \lstinline|v[mid] < x|.
  \end{uncoverenv}
\end{frame}

\section{Variații în STL}

\begin{frame}[fragile]{STL: \texttt{binary\_search}, \texttt{lower\_bound}, \texttt{upper\_bound}}
  C++ oferă aceste funcții gata implementate în \lstinline|<algorithm>|, toate cu complexitate $O(\log n)$.
  \begin{intrebare}
    Cum se află, cu aceste funcții, dacă $x$ există în șir, prima lui poziție și numărul lui de apariții?
  \end{intrebare}
  \begin{uncoverenv}<2->
    \begin{lstlisting}
#include <algorithm>
using namespace std;

int v[] = {1, 2, 2, 2, 3, 5};
int n = 6;

// Exista x in v? (true/false)
bool exista = binary_search(v, v + n, 2);

// Pozitii (indici)
int poz1 = lower_bound(v, v + n, 2) - v;   // 1: prima aparitie
int poz2 = upper_bound(v, v + n, 2) - v;   // 4: dupa ultima

// Numarul de aparitii ale lui x
int cate = poz2 - poz1;                    // 3
    \end{lstlisting}
  \end{uncoverenv}
  \beginnote{\texttt{lower\_bound} întoarce un pointer / iterator; scăzând începutul tabloului se obține indicele}
\end{frame}

\begin{frame}[fragile]{STL \textemdash{} numărare pe interval $[x, y]$}
  \begin{intrebare}
    Cum se numără elementele unui șir care se află în intervalul $[x, y]$?
  \end{intrebare}
  \begin{hint}<2->
    Șirul se sortează, apoi se folosesc \texttt{lower\_bound} pentru $x$ și \texttt{upper\_bound} pentru $y$.
  \end{hint}
  \begin{uncoverenv}<3->
    \begin{lstlisting}
int v[] = {23, 2, 5, 26, 11, 17, 14, 8, 20};
int n = 9;
sort(v, v + n);                  // sirul trebuie sortat inainte

int x = 8, y = 20;
int st = lower_bound(v, v + n, x) - v;   // prima pozitie cu v[poz] >= x
int dr = upper_bound(v, v + n, y) - v;   // prima pozitie cu v[poz] > y

int cate = dr - st;                      // 5: 8, 11, 14, 17, 20
    \end{lstlisting}
  \end{uncoverenv}
  \begin{itemize}
    \item<4-> Idee frecventă: interogări pe un șir sortat în $O(\log n)$ în loc de $O(n)$
  \end{itemize}
\end{frame}

\section{Complexitate}

\begin{frame}{Complexitatea căutării binare}
  La fiecare pas, intervalul $[st, dr]$ se \textbf{înjumătățește}:
  \[
    n \;\to\; \frac{n}{2} \;\to\; \frac{n}{4} \;\to\; \cdots \;\to\; 1
  \]
  \begin{intrebare}
    După câți pași ajunge intervalul la un singur element?
  \end{intrebare}
  \uncover<2->{Numărul de pași este $\log_2 n$.}
  \uncover<3->{
    \begin{center}
    \begin{tabular}{lcc}
      \toprule
      $n$ & $\log_2 n$ & Pași maximi \\
      \midrule
      $10$              & $3{,}32$   & $4$  \\
      $1\,000$          & $9{,}97$   & $10$ \\
      $10^6$            & $19{,}93$  & $20$ \\
      $10^9$            & $29{,}90$  & $30$ \\
      $10^{18}$         & $59{,}79$  & $60$ \\
      \bottomrule
    \end{tabular}
    \end{center}}
  \begin{remarca}<4->{Concluzie}
    Chiar și pentru $n = 10^{18}$ sunt necesari cel mult $60$ de pași.
  \end{remarca}
\end{frame}

\section{Căutare binară pe rezultat}

\begin{frame}{Căutare binară pe rezultat \textemdash{} idee}
  \begin{itemize}
    \item Căutarea binară clasică caută o \textbf{poziție} într-un tablou sortat
    \item Căutarea binară \textbf{pe rezultat} caută direct \textbf{valoarea răspunsului}, fără ca tabloul să existe
  \end{itemize}
  \begin{intrebare}
    Ce informații sunt necesare pentru a căuta binar valoarea unui răspuns?
  \end{intrebare}
  \begin{block}<2->{Schema în trei pași}
    \begin{enumerate}
      \item Se stabilește intervalul $[st, dr]$ în care se află răspunsul
      \item Se scrie o funcție $\mathrm{check}(x)$: „se poate rezolva problema cu valoarea $x$?”
      \item Se verifică faptul că $\mathrm{check}$ este \textbf{monotonă}, apoi se caută pragul: cel mai mare $x$ cu $\mathrm{check}(x)$ adevărat (sau cel mai mic, în varianta simetrică)
    \end{enumerate}
  \end{block}
  \begin{itemize}
    \item<3-> Cuvinte-cheie în enunț: „cel mai mic / cel mai mare $x$ astfel încât\ldots”, „minimizează maximul”, „maximizează minimul”
  \end{itemize}
  \beginnote{Un număr din $[1, 10^9]$ se ghicește în cel mult $30$ de întrebări „mai mare sau mai mic?”, deoarece $2^{30} > 10^9$}
\end{frame}

\begin{frame}{Căutare binară pe rezultat \textemdash{} desfășurare}
  Se caută cel mai mare $x$ cu $\mathrm{check}(x)$ adevărat; aici $\mathrm{check}(x)$ este adevărat exact pentru $x \le 10$. Intervalul necercetat $[st, dr]$ începe cu $[0, 16]$; celula cu margine albastră este $mid$.
  \begin{intrebare}
    Cum se modifică $st$, $dr$ și $ans$ după fiecare test?
  \end{intrebare}
  \uncover<2->{Dacă $\mathrm{check}(mid)$ este adevărat, $ans = mid$ și $st = mid + 1$; altfel $dr = mid - 1$.}
  \smallskip

  \centering
  \begin{tikzpicture}
    \uncover<3->{\bsrow{0}{0}{16}{8}{$mid=8$: adevărat $\rightarrow$ $ans=8$, $st=9$}}
    \uncover<4->{\bsrow{1}{9}{16}{12}{$mid=12$: fals $\rightarrow$ $dr=11$}}
    \uncover<5->{\bsrow{2}{9}{11}{10}{$mid=10$: adevărat $\rightarrow$ $ans=10$, $st=11$}}
    \uncover<6->{\bsrow{3}{11}{11}{11}{$mid=11$: fals $\rightarrow$ $dr=10$}}
    \uncover<7->{\bsrow{4}{11}{10}{-1}{$st > dr$: răspuns $ans=10$}}
  \end{tikzpicture}

  \raggedright
  \uncover<7->{\textcolor{igGreen}{$\blacksquare$} adevărat (cunoscut) \quad \textcolor{igRed}{$\blacksquare$} fals (cunoscut) \quad $\square$ necunoscut\\
  Aici sunt $4$ teste pentru $17$ valori; pentru un domeniu de $10^{18}$ valori, circa $60$ de teste.}
\end{frame}

\begin{frame}{Căutare binară pe rezultat \textemdash{} exemplu}
  \begin{block}{Problemă}
    Se dau $n$ bețe de lungimi $d_1, d_2, \dots, d_n$. Din fiecare băț se taie bucăți de aceeași lungime întreagă $L$ (restul rămas se aruncă). Să se determine \textbf{lungimea maximă} $L$ pentru care se obțin în total \textbf{cel puțin} $A$ bucăți.
  \end{block}
  Restricții: $1 \leq n \leq 10^5$, $1 \leq d_i \leq 10^9$, $1 \leq A \leq 10^9$.

  Exemplu: $d = [7,\ 4,\ 9,\ 5]$, $A = 6$ $\Rightarrow$ $L = 3$ (se obțin $2+1+3+1 = 7 \geq 6$ bucăți).
  \begin{intrebare}<2->
    Cum se poate rezolva problema dacă se încearcă pe rând fiecare lungime posibilă, de la cea mai mare la cea mai mică?
  \end{intrebare}
  \uncover<3->{Pentru fiecare $L$ se calculează $\sum_i \lfloor d_i / L \rfloor$ și se oprește la prima valoare a lui $L$ pentru care suma este $\geq A$. Complexitate $O(n \cdot maxim)$, unde $maxim$ este lungimea maximă a unui băț: pentru $maxim = 10^9$ și $n = 10^5$, mult prea lent.}
\end{frame}

\begin{frame}{Căutare binară pe rezultat \textemdash{} observație}
  Pe măsură ce $L$ crește, numărul de bucăți obținute \textbf{scade} (sau rămâne la fel).

  Exemplu: $d = [7,\ 4,\ 9,\ 5]$, $A = 6$
  \begin{center}
  \begin{tabular}{lccccccccc}
    \toprule
    $L$                & 1  & 2  & 3 & 4 & 5 & 6 & 7 & 8 & 9 \\
    \midrule
    $\mathrm{bucati}(L)$ & 25 & 11 & 7 & 5 & 3 & 2 & 2 & 1 & 1 \\
    $\geq A$?          & \at{2}{\checkmark} & \at{2}{\checkmark} & \at{2}{\checkmark} & \at{2}{$\times$} & \at{2}{$\times$} & \at{2}{$\times$} & \at{2}{$\times$} & \at{2}{$\times$} & \at{2}{$\times$} \\
    \bottomrule
  \end{tabular}
  \end{center}
  \begin{intrebare}<3->
    Ce proprietate are șirul valorilor $\mathrm{check}(L)$, de la $L = 1$ spre dreapta?
  \end{intrebare}
  \begin{remarca}<4->{Monotonie}
    Odată ce numărul de bucăți scade sub $A$, rămâne sub $A$. Condiția trece o singură dată de la adevărat la fals, deci la fiecare test se poate elimina jumătate din candidați.
  \end{remarca}
\end{frame}

\begin{frame}[fragile]{Căutare binară pe rezultat \textemdash{} implementare}
  \begin{lstlisting}
long long bucati(int d[], int n, int L) {
    long long cnt = 0;
    for (int i = 0; i < n; i++) {
        cnt += d[i] / L;
    }
    return cnt;
}

int st = 1, dr = maxim, ans = 0;     // maxim = cel mai mare d[i]
while (st <= dr) {
    int mid = (st + dr) / 2;
    if (bucati(d, n, mid) >= A) {
        ans = mid;         // functioneaza, incercam mai mare
        st = mid + 1;
    } else {
        dr = mid - 1;      // nu functioneaza, incercam mai mic
    }
}
cout << ans;
  \end{lstlisting}
  \begin{itemize}
    \item Complexitate: $O(n \log(maxim))$ \textemdash{} pentru aceleași date, $\approx 10^5 \cdot 30 = 3 \cdot 10^6$ operații
  \end{itemize}
\end{frame}

\begin{frame}{Overflow și condiția de aplicabilitate}
  \begin{intrebare}
    De ce poate apărea overflow la $mid = (st + dr) \ / \ 2$ în căutarea binară pe rezultat?
  \end{intrebare}
  \begin{block}<2->{Overflow}
    Intervalul $[st, dr]$ poate fi foarte mare, iar suma $st + dr$ poate depăși limita tipului \lstinline|int| ($\approx 2{,}1 \cdot 10^9$). Se folosește \lstinline|long long| sau \lstinline[keepspaces=true]|mid = st + (dr - st) / 2|. La căutarea unui element, $[st, dr]$ este mărginit de $n$, deci riscul nu apare de obicei.
  \end{block}
  \begin{block}<3->{Condiția de aplicabilitate}
    Trebuie să existe un \textbf{singur prag} în care $\mathrm{check}$ trece din adevărat în fals (sau invers). Verificare practică: dacă $L$ este valid, este valid și $L - 1$?
    \begin{itemize}
      \item $\checkmark\ \checkmark\ \checkmark\ \times\ \times\ \times$ \textemdash{} monotonă, se poate aplica
      \item $\checkmark\ \times\ \checkmark\ \times\ \checkmark$ \textemdash{} nu este monotonă, nu se poate aplica
    \end{itemize}
  \end{block}
\end{frame}

\section{Recapitulare}

\begin{frame}{Recapitulare}
  \begin{center}
  \begin{tabular}{lll}
    \toprule
    Variantă & Timp & Spațiu \\
    \midrule
    Căutare secvențială             & $O(n)$      & $O(1)$ \\
    Căutare binară (element)        & $O(\log n)$ & $O(1)$ \\
    STL \texttt{lower/upper\_bound} & $O(\log n)$ & $O(1)$ \\
    Căutare binară pe rezultat      & $O(\log V \cdot \mathrm{cost}(\mathrm{check}))$ & depinde \\
    \bottomrule
  \end{tabular}
  \end{center}
  \begin{itemize}
    \item Șirul trebuie sortat înainte de căutarea binară a unui element
    \item Pentru intervale $[st, dr]$ foarte mari se folosește \lstinline|long long| sau \lstinline[keepspaces=true]|mid = st + (dr - st) / 2|
    \item \texttt{lower\_bound} $\rightarrow$ prima poziție cu $\geq x$; \texttt{upper\_bound} $\rightarrow$ prima poziție cu $> x$
    \item La căutarea binară pe rezultat, $\mathrm{check}$ trebuie să fie monotonă; complexitatea este $O(\log V \cdot \mathrm{cost}(\mathrm{check}))$, unde $V = dr - st$
  \end{itemize}
\end{frame}
